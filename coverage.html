// ProtoToEntity convierte un PlanEntity (gRPC)
// en una entidad de MongoDB.

// ----------------------------------------------------------------------
// Proto -> Entity
// ----------------------------------------------------------------------

func ProtoToEntity(proto *planpb.PlanEntity) (*entity.Plan, error) {

        var objectID bson.ObjectID

        if proto.GetId() != "" {
                id, err := bson.ObjectIDFromHex(proto.GetId())
                if err != nil {
                        return nil, err
                }
                objectID = id
        }

        var idCompany bson.ObjectID

        if proto.GetPlanIdCompany() != "" {
                id, err := bson.ObjectIDFromHex(proto.GetPlanIdCompany())
                if err != nil {
                        return nil, err
                }
                idCompany = id
        }

        plan := &entity.Plan{
                ID:                     objectID,
                PlanIDCompany:          idCompany,
                PlanAccentColor:        proto.GetPlanAccentColor(),
                PlanAccentTextColor:    proto.GetPlanAccentTextColor(),
                PlanTitle:              proto.GetPlanTitle(),
                PlanSubtitle:           proto.GetPlanSubtitle(),
                PlanDescriptionTinyMCE: proto.GetPlanDescriptionTinymce(),
                PlanType:               proto.GetPlanType(),
                PlanCurrencyPrice:      proto.GetPlanCurrencyPrice(),
                PlanMenu:               proto.GetPlanMenu(),
                PlanDescriptionItems:   proto.GetPlanDescriptionItems(),

                PlanDays:       proto.GetPlanDays(),
                PlanOffices:    proto.GetPlanOffices(),
                PlanUsers:      proto.GetPlanUsers(),
                PlanTotalSales: proto.GetPlanTotalSales(),
                PlanPrice:      proto.GetPlanPrice(),

                PlanFeatured: proto.GetPlanFeatured(),
                PlanCustom:   proto.GetPlanCustom(),
                ActivePlan:   proto.GetActivePlan(),
        }

        if proto.GetCreatedAt() != nil {
                plan.CreatedAt = proto.GetCreatedAt().AsTime()
        }

        if proto.GetUpdatedAt() != nil {
                plan.UpdatedAt = proto.GetUpdatedAt().AsTime()
        }

        if proto.GetDeletedAt() != nil {
                t := proto.GetDeletedAt().AsTime()
                plan.DeletedAt = &t
        }

        return plan, nil
}

// EntityToProto convierte una entidad MongoDB
// en PlanEntity (gRPC).

// ----------------------------------------------------------------------
// Entity -> Proto
// ----------------------------------------------------------------------

func EntityToProto(plan *entity.Plan) *planpb.PlanEntity {
        if plan == nil {
                return nil
        }

        var id string
        if !plan.ID.IsZero() {
                id = plan.ID.Hex()
        }

        var planIDCompany string
        if !plan.PlanIDCompany.IsZero() {
                planIDCompany = plan.PlanIDCompany.Hex()
        }

        var deletedAt *timestamppb.Timestamp
        if plan.DeletedAt != nil {
                deletedAt = timestamppb.New(*plan.DeletedAt)
        }

        return &planpb.PlanEntity{
                Id:                     id,
                PlanIdCompany:          planIDCompany,
                PlanAccentColor:        plan.PlanAccentColor,
                PlanAccentTextColor:    plan.PlanAccentTextColor,
                PlanTitle:              plan.PlanTitle,
                PlanSubtitle:           plan.PlanSubtitle,
                PlanDescriptionTinymce: plan.PlanDescriptionTinyMCE,
                PlanType:               plan.PlanType,
                PlanCurrencyPrice:      plan.PlanCurrencyPrice,
                PlanMenu:               plan.PlanMenu,
                PlanDescriptionItems:   plan.PlanDescriptionItems,

                PlanDays:       plan.PlanDays,
                PlanOffices:    plan.PlanOffices,
                PlanUsers:      plan.PlanUsers,
                PlanTotalSales: plan.PlanTotalSales,
                PlanPrice:      plan.PlanPrice,

                PlanFeatured: plan.PlanFeatured,
                PlanCustom:   plan.PlanCustom,
                ActivePlan:   plan.ActivePlan,

                CreatedAt: timestamppb.New(plan.CreatedAt),
                UpdatedAt: timestamppb.New(plan.UpdatedAt),
                DeletedAt: deletedAt,
        }
}

// EntitiesToProto convierte una lista de entidades
// en una lista de PlanEntity.
func EntitiesToProto(bases []*entity.Plan) []*planpb.PlanEntity {

        result := make([]*planpb.PlanEntity, 0, len(bases))

        for _, plan := range bases {
                result = append(result, EntityToProto(plan))
        }

        return result
}

// applyCreateDefaults aplica los valores por defecto para una creación.
func (s *BaseService) applyCreateDefaults(
        entity *entity.Plan,
) {

        if entity == nil {
                return
        }

        now := time.Now().UTC()

        // MongoDB genera automáticamente el _id,
        // pero si queremos comportarnos como Mongoose
        // lo generamos antes de guardar.
        if entity.ID.IsZero() {
                entity.ID = bson.NewObjectID()
        }

        if entity.CreatedAt.IsZero() {
                entity.CreatedAt = now
        }

        entity.UpdatedAt = now
        entity.DeletedAt = nil

        // Valor por defecto.
        entity.PlanTotalSales = 0
}

// applyUpdateDefaults aplica los valores por defecto
// antes de actualizar un documento.
func (s *BaseService) applyUpdateDefaults(
        update bson.M,
) {

        if update == nil {
                return
        }

        update["updatedAt"] = time.Now().UTC()

        delete(update, "_id")
        delete(update, "createdAt")
        delete(update, "deletedAt")
}

// Construye una llave de caché versionada para el namespace de la entidad.
func (s *BaseService) cacheKey(
        ctx context.Context,
        tenantNamespace []any, // Ej: []any{"company", idCompany}
        parts ...any, // Ej: "one", id.Hex()

) (string, error) {

        // 1. Obtenemos la versión ESPECÍFICA de este tenant
        version, err := s.cacheVersion(ctx, tenantNamespace...)

// Create registra una nueva entidad.
func (s *BaseService) Create(
        ctx context.Context,
        plan *entity.Plan,

) error {

        s.applyCreateDefaults(plan)

        //----------------------------------------------------------------------
        // EJECUCIÓN PARALELA
        //----------------------------------------------------------------------

        g, ctxGroup := errgroup.WithContext(ctx)

        // ------------------------------------------------------------------
        // VALIDACIONES
        // ------------------------------------------------------------------

        // Validaciones locales.
        g.Go(func() error {

                if plan.PlanTitle == "" {
                        return status.Error(
                                codes.InvalidArgument,
                                "Plan title is required",
                        )
                }

                if plan.PlanSubtitle == "" {
                        return status.Error(
                                codes.InvalidArgument,
                                "Plan subtitle is required",
                        )
                }

                return nil
        })

        // Verificar que el plan no exista.
        g.Go(func() error {

                existing, err := s.FindByField(
                        ctxGroup,
                        "plan_title",
                        plan.PlanTitle,
                )

                if err != nil {

                        // Que no exista el plan NO es un error.
                        if errors.Is(err, mongo.ErrNoDocuments) {
                                return nil
                        }

                        return err
                }

                if existing != nil {
                        return status.Error(
                                codes.AlreadyExists,
                                "The plan is already registered",
                        )
                }

                return nil
        })

        if err := g.Wait(); err != nil {
                return err
        }

        //----------------------------------------------------------------------
        // REGLA DE NEGOCIO
        //----------------------------------------------------------------------

        // Si el nuevo plan será destacado,
        // todos los demás planes destacados dejan de serlo.
        if plan.PlanFeatured {

                if err := s.repository.UpdateMany(

                                "plan_featured": true,

                                "plan_featured": false,
                        },

                ); err != nil {

                        logger.Error(err.Error())

                        return status.Error(
                                codes.Internal,
                                "internal server error",
                        )
                }
        }

        //----------------------------------------------------------------------
        // PERSISTENCIA
        //----------------------------------------------------------------------

        if err := s.repository.Create(ctx, plan); err != nil {

                if mongo.IsDuplicateKeyError(err) {
                        return status.Error(
                                codes.AlreadyExists,
                                "The plan is already registered",
                        )
                }

                logger.Error(err.Error())

                return status.Error(
                        codes.Internal,
                        "internal server error",
                )
        }

        //----------------------------------------------------------------------
        // INVALIDAR CACHÉ
        //----------------------------------------------------------------------

        if err := s.bumpCacheVersion(
                ctx,
                "plan",
                "global",
        
        ); err != nil {
                logger.Error(err.Error())
        }

        //----------------------------------------------------------------------
        // PUBLICAR EVENTO
        //----------------------------------------------------------------------

        event := pulsar.NewEvent(
                pulsar.TopicPlanCreated,
                "ms-plan",
                plan,
        )

        event.WithCorrelation(plan.ID.Hex())

        if err := pulsar.Publish(

                pulsar.TopicPlanCreated,
                event,
        
        ); err != nil {

                logger.Error(err.Error())

                // return status.Error(
                //         codes.Internal,
                //         "internal server error",
                // )
        } else {

                logger.Info(
                        "Evento publicado",
                        "topic", pulsar.TopicPlanCreated,
                        "entity_id", plan.ID.Hex(),
                )
        }

        return nil
}

// Hash genera un hash SHA-256 determinístico de cualquier estructura.
//
// Se utiliza principalmente para construir llaves de caché cuando una
// consulta depende de filtros, paginación o parámetros complejos.
//
// Ejemplo:
//
//        filter := bson.M{
//            "status": true,
//            "company": companyID,
//        }
//
//        hash, _ := cache.Hash(filter)
//
//        key := cache.Key("color", "list", hash)
func Hash(v any) (string, error) {

        data, err := json.Marshal(v)
        if err != nil {
                return "", err
        }

        sum := sha256.Sum256(data)

        return hex.EncodeToString(sum[:]), nil
}

package cache

import (
        "fmt"
        "strings"
)

// Separator utilizado para construir todas las llaves.
const Separator = ":"

// Key construye una llave uniendo todas sus partes.
//
// Ejemplos:
//
//        Key("color", "id", id)
//        => color:id:685b6c2f...
//
//        Key("invoice", "page", hash)
//        => invoice:page:a87dc12...
func Key(parts ...any) string {

        values := make([]string, 0, len(parts))

        for _, part := range parts {
                values = append(values, fmt.Sprint(part))
        }

        return strings.Join(values, Separator)
}

// VersionKey devuelve la llave donde se almacena la versión
// de un namespace.
//
// Ejemplo:
//
//        VersionKey("color", "company", companyID)
//
// Resultado:
//
//        color:company:10:version
func VersionKey(parts ...any) string {

        values := append(parts, "version")

        return Key(values...)
}

// CacheKey devuelve una llave versionada.
//
// Ejemplo:
//
//        CacheKey(
//            version,
//            "color",
//            "company",
//            companyID,
//            "list",
//            hash,
//        )
//
// Resultado:
//
//        color:company:10:v:15:list:af87c...
func CacheKey(
        version int64,
        parts ...any,
) string {

        values := make([]any, 0, len(parts)+2)

        values = append(values, parts...)
        values = append(values, "v")
        values = append(values, version)

        return Key(values...)
}

// Subscribe crea un Consumer para un tópico y comienza
// a escuchar mensajes en una goroutine.
func Subscribe(
        topic string,
        subscription string,
        handler Handler,
        retryCfg RetryConfig,
        store IdempotencyStore,

) error {

        if handler == nil {
                return fmt.Errorf("handler cannot be nil")
        }

        consumer, err := Client().Subscribe(
                pulsarclient.ConsumerOptions{
                        Topic:            topic,
                        SubscriptionName: subscription,
                        Type:             pulsarclient.Shared,
                },
        )

        if err != nil {
                return fmt.Errorf("error creating consumer: %w", err)
        }

        mutex.Lock()
        consumers = append(consumers, consumer)
        mutex.Unlock()

        go consume(consumer, topic, handler, retryCfg, store)

        return nil
}

        mutex.RLock()
        defer mutex.RUnlock()

        result := make([]pulsarclient.Consumer, len(consumers))
        copy(result, consumers)

        return result
}

                                ackOrRequeue(consumer, msg, topic)
                                continue
                        }

                        }

                // 5. ACK
                //------------------------------------------------------------

                // consumer.Ack(msg)

                ackOrRequeue(consumer, msg, topic)

// Este archivo encapsula la lógica para enviar a la Dead Letter Queue los mensajes que fallaron definitivamente después de agotar los reintentos.

// DLQEnvelope empaqueta el evento original junto a la metadata de seguridad y error.
type DLQEnvelope struct {
        OriginalTopic string    `json:"original_topic"`
        ErrorReason   string    `json:"error_reason"`
        FailedAt      time.Time `json:"failed_at"`
        Payload       []byte    `json:"payload"`
}

// SendToDLQ redirige un payload irrecuperable a la cola de mensajes muertos.
func SendToDLQ(ctx context.Context, originalTopic string, payload []byte, reason error) error {
        dlqTopic := fmt.Sprintf("%s.dlq", originalTopic)

        envelope := DLQEnvelope{
                OriginalTopic: originalTopic,
                ErrorReason:   reason.Error(),
                FailedAt:      time.Now().UTC(),
                Payload:       payload,
        }

        // Reutilizamos el pool de Producers de producer.go
        if err := Publish(ctx, dlqTopic, envelope); err != nil {
                return fmt.Errorf("failed to route message to DLQ topic %s: %w", dlqTopic, err)
        }

        return nil
}

var (
        producers     = make(map[string]pulsarclient.Producer)
        producerMutex sync.RWMutex
)

// Producer retorna (o crea) un Producer para un tópico.
// Cada tópico mantiene un único Producer reutilizable.
func Producer(topic string) (pulsarclient.Producer, error) {

        // Buscar si ya existe.
        producerMutex.RLock()

        producer, ok := producers[topic]

        producerMutex.RUnlock()

        if ok {
                return producer, nil
        }

        // Crear uno nuevo.
        producerMutex.Lock()
        defer producerMutex.Unlock()

        // Doble verificación por concurrencia.
        if producer, ok := producers[topic]; ok {
                return producer, nil
        }

        var err error

        producer, err = Client().CreateProducer(
                pulsarclient.ProducerOptions{
                        Topic: topic,
                },
        )

        if err != nil {
                return nil, fmt.Errorf("failed to create producer: %w", err)
        }

        producers[topic] = producer

        return producer, nil
}

// Publish serializa un evento y lo publica en el tópico indicado.
func Publish(
        ctx context.Context,
        topic string,
        event any,

) error {

        producer, err := Producer(topic)
        if err != nil {
                return err
        }

        data, err := json.Marshal(event)
        if err != nil {
                return fmt.Errorf("failed to marshal event: %w", err)
        }

        // Evita que una publicación quede bloqueada indefinidamente.
        ctx, cancel := context.WithTimeout(ctx, 5*time.Second)
        defer cancel()

        _, err = producer.Send(
                ctx,
                &pulsarclient.ProducerMessage{
                        Payload: data,
                },
        )

        if err != nil {
                return fmt.Errorf("failed to publish event: %w", err)
        }

        return nil
}

// Este archivo maneja la tolerancia a fallos mediante reintentos con Exponential Backoff y Jitter
// (variación aleatoria) para no saturar los servicios en caso de caídas intermitentes.

// RetryConfig define la configuración para la estrategia de reintentos.
type RetryConfig struct {
        MaxRetries   int           // Número máximo de intentos
        InitialDelay time.Duration // Tiempo base de espera inicial
        MaxDelay     time.Duration // Tiempo máximo de espera
        Factor       float64       // Multiplicador exponencial
}

// DefaultRetryConfig entrega una configuración robusta por defecto.
func DefaultRetryConfig() RetryConfig {
        return RetryConfig{
                MaxRetries:   3,
                InitialDelay: 200 * time.Millisecond,
                MaxDelay:     3 * time.Second,
                Factor:       2.0,
        }
}

// ExecuteWithRetry ejecuta un Handler envolviéndolo en una política de reintentos.
func ExecuteWithRetry(ctx context.Context, cfg RetryConfig, handler Handler, payload []byte) error {
        var err error

        for attempt := 0; attempt <= cfg.MaxRetries; attempt++ {
                // Verificar si el contexto fue cancelado desde arriba (ej. shutdown)
                if err := ctx.Err(); err != nil {
                        return fmt.Errorf("retry canceled by context: %w", err)
                }

                err = handler(ctx, payload)
                if err == nil {
                        return nil // Ejecución exitosa, salimos del ciclo
                }

                // Si es el último intento, no esperamos, simplemente rompemos
                if attempt == cfg.MaxRetries {
                        break
                }

                // Cálculo de Exponential Backoff
                delay := float64(cfg.InitialDelay) * math.Pow(cfg.Factor, float64(attempt))
                if delay > float64(cfg.MaxDelay) {
                        delay = float64(cfg.MaxDelay)
                }

                // Agregar Jitter (variación de +/- 20%) para evitar el efecto manada (thundering herd)
                jitter := (rand.Float64()*0.4 - 0.2) * delay
                sleepDuration := time.Duration(delay + jitter)

                select {
                case <-time.After(sleepDuration): // Esperamos antes del próximo reintento
                case <-ctx.Done():
                        return ctx.Err()
                }
        }

        return fmt.Errorf("exceeded max retries (%d): %w", cfg.MaxRetries, err)
}

// Close libera todos los recursos utilizados por Apache Pulsar.
func Close() {

        //---------------------------------------------------
        // Cerrar Consumers
        //---------------------------------------------------

        mutex.Lock()

        for _, consumer := range consumers {

                if consumer != nil {
                        consumer.Close()
                }
        }

        consumers = nil

        mutex.Unlock()

        //---------------------------------------------------
        // Cerrar Producers
        //---------------------------------------------------

        producerMutex.Lock()

        for topic, producer := range producers {

                if producer != nil {
                        producer.Close()
                }

                delete(producers, topic)
        }

        producerMutex.Unlock()

        //---------------------------------------------------
        // Cerrar Client
        //---------------------------------------------------

        if client != nil {
                client.Close()
                client = nil
        }
}

// Reset elimina el estado interno.
// Útil para pruebas unitarias.
func Reset() {

        mutex.Lock()
        consumers = nil
        mutex.Unlock()

        producerMutex.Lock()
        producers = make(map[string]pulsarclient.Producer)
        producerMutex.Unlock()

        client = nil

        once = sync.Once{}
}
